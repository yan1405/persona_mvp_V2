```jsx
<Input variant="search" placeholder="Buscar registros, evidências e competências" />
<Input label="Vaga ou objetivo" hint="Usado para recomendar evidências" />
<Input label="Link de prova" error="Link inválido. Use um endereço que comece com http:// ou https://" />
```
